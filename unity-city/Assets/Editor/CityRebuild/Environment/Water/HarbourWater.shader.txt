Shader "Braxhorn/HarbourWaterMobile"
{
    // MIASTO R1 18.11 (właściciel: „nie możemy zrobić bardziej realistycznej wody?”, „woda… nie pasek… widać nic”):
    // woda jak prawdziwa — fale wiatrowe z widma (mapa normalnych FFT, dwie warstwy), odbicie TEGO SAMEGO zdjęcia 360° co
    // niebo miasta (Fresnel n = 1.33), pod powierzchnią: światło po drodze w wodzie (pochłanianie R/G/B na metr), dno z piasku
    // widoczne przez płytką wodę z załamaniem (prawo Snella) i kaustykami, głęboka woda ciemnieje; odblask słońca.
    // Nadal jeden nieprzezroczysty przebieg (bez grab pass / tekstury głębi kamery) — koszt telefonu jak dotąd + kilka próbek.
    Properties
    {
        _DeepColor ("Light scattered by deep water (linear)", Color) = (0.010,0.052,0.070,1)
        _SkyHorizon ("Sky horizon if no panorama and no reflection", Color) = (0.43,0.49,0.52,1)
        _SkyZenith ("Sky zenith if no panorama and no reflection", Color) = (0.28,0.40,0.53,1)
        _SunDirection ("Direction towards daylight if no realtime light", Vector) = (-0.45,0.78,-0.42,0)
        _NearRectangle ("Moving mesh x0 z0 x1 z1", Vector) = (84,-120,560,260)
        _ReflectionStrength ("Reflection contribution", Range(0,1)) = 1
        _ReflectionMip ("Cubemap reflection softness (no panorama)", Range(0,6)) = 1.5
        _RippleStrength ("Wind wave slope", Range(0,1.5)) = 1
        _SwellStrength ("Broad wave amplitude scale", Range(0,1)) = 1
        [NoScaleOffset] _NieboTex ("Sky panorama (equirectangular, as the city sky)", 2D) = "black" {}
        _NieboObrot ("Sky panorama rotation (radians)", Float) = 0
        _NieboEkspozycja ("Sky panorama exposure", Float) = 1
        _NieboJest ("Sky panorama bound (0/1)", Float) = 0
        [NoScaleOffset] _FaleNormal ("Wind wave normals (R=x G=z B=up)", 2D) = "bump" {}
        _FaleSkala ("Wind wave tile (m)", Float) = 10
        [NoScaleOffset] _Kaustyki ("Seabed caustics (sqrt(I/4))", 2D) = "gray" {}
        _KaustykiSkala ("Caustics tile (m)", Float) = 3
        [NoScaleOffset] _DnoTex ("Water depth (sqrt(depth/max))", 2D) = "white" {}
        _DnoProstokat ("Depth texture x0 z0 x1 z1", Vector) = (60,-452,1468,572)
        _DnoMax ("Depth texture max (m)", Float) = 20
        [NoScaleOffset] _DnoAlbedo ("Seabed sand", 2D) = "gray" {}
        _DnoSkala ("Seabed sand tile (m)", Float) = 2.4
        _DnoKolor ("Seabed sand tint", Color) = (0.80,0.75,0.67,1)
        _Pochlanianie ("Light attenuation per metre (RGB)", Vector) = (0.45,0.09,0.075,0)
        _SlonceKolor ("Sun colour x intensity if no realtime light", Color) = (1.24,1.184,1.085,1)
        _Otoczenie ("Sky ambient if the scene has none", Color) = (0.60,0.68,0.80,1)
        _DalekoKolor ("Far sea of the sky shader (photo mode)", Color) = (0.64,0.70,0.75,1)
        _DalekoOd ("Far sea blend start (m)", Float) = 700
        _DalekoDo ("Far sea blend end (m, 0 = off)", Float) = 0
    }
    SubShader
    {
        Tags { "RenderType"="Opaque" "Queue"="Geometry+10" "IgnoreProjector"="True" }
        LOD 150
        CGINCLUDE
            #include "UnityCG.cginc"
            #include "UnityLightingCommon.cginc"
            // MIASTO R1 — doba (24.11, właściciel: „noc i dzień”): poziom i barwa światła dnia z BraxObrazRejonu (jak
            // Assets/Claude/Shaders/BraxSwiatlo.cginc — tu wpisane, bo shader jest kopiowany do PlayableGenerated); 0 = dzień
            float _BraxSwiatlo; half4 _BraxSwiatloKolor;
            inline half3 BraxDobaWody(half3 c) { half s = _BraxSwiatlo > 0 ? _BraxSwiatlo : 1; return c * (_BraxSwiatloKolor.a > 0 ? _BraxSwiatloKolor.rgb : half3(1, 1, 1)) * s; }

            #define W_PI 3.14159265
            #define W_2PI 6.28318531

            half4 _DeepColor, _SkyHorizon, _SkyZenith, _DnoKolor, _Pochlanianie, _SlonceKolor, _Otoczenie, _DalekoKolor;
            float _DalekoOd, _DalekoDo;
            float4 _SunDirection, _NearRectangle, _DnoProstokat;
            half _ReflectionStrength, _ReflectionMip, _RippleStrength, _SwellStrength, _NieboEkspozycja, _NieboJest;
            float _NieboObrot, _FaleSkala, _KaustykiSkala, _DnoMax, _DnoSkala;
            sampler2D _NieboTex, _FaleNormal, _Kaustyki, _DnoTex, _DnoAlbedo;

            struct appdata { float4 vertex : POSITION; };
            struct v2f
            {
                float4 position : SV_POSITION;
                float3 world : TEXCOORD0;
                UNITY_FOG_COORDS(1)
            };

            // The mean surface remains at -1.7 m. Displacement fades to zero
            // at every edge of the fine mesh, so adjoining flat patches meet.
            float3 SwellEnvelope(float2 p)
            {
                float4 widths = float4(1.5,16.0,16.0,16.0);
                float4 t = saturate((float4(p.x,p.x,p.y,p.y)
                         - float4(_NearRectangle.x,_NearRectangle.z-16.0,
                                  _NearRectangle.y,_NearRectangle.w-16.0))/widths);
                float4 a = t*t*(3.0-2.0*t);
                float4 d = 6.0*t*(1.0-t)/widths;
                float value = a.x*(1.0-a.y)*a.z*(1.0-a.w);
                float dx = (d.x*(1.0-a.y)-a.x*d.y)*a.z*(1.0-a.w);
                float dz = a.x*(1.0-a.y)*(d.z*(1.0-a.w)-a.z*d.w);
                return float3(value,dx,dz)*_SwellStrength;
            }

            float2 SwellPhase(float2 p)
            {
                float2 q = p - float2(84.0,60.0);
                return float2(dot(q,float2(0.84,0.5426))*0.84,
                              dot(q,float2(-0.4191,0.9079))*1.46)
                       - _Time.y * float2(0.65,0.92);
            }

            // Odbite niebo: to samo zdjęcie 360° i ten sam obrót co niebo miasta (MiastoTlo.cginc, MT_NieboUV), z rozmiarem
            // piksela odbicia (tex2Dgrad, bez szwu atan2). Bez panoramy: dawna mapa odbić sceny albo gradient nieba.
            half3 Zdjecie(float3 r, float kat)
            {
                float lon = atan2(r.z, r.x) - _NieboObrot;
                float lat = asin(clamp(r.y, -1.0, 1.0));
                float2 uv = float2(frac(0.5 - lon / W_2PI), 0.5 + lat / W_PI);
                float c = max(sqrt(saturate(1.0 - r.y * r.y)), 0.03);
                return tex2Dgrad(_NieboTex, uv, float2(kat / (W_2PI * c), 0.0), float2(0.0, kat / W_PI)).rgb * _NieboEkspozycja;
            }

            half3 Niebo(float3 r, float kat)
            {
                r.y = max(r.y, 0.004);
                r = normalize(r);
                if (_NieboJest > 0.5) return Zdjecie(r, kat);
                half4 encodedSky = UNITY_SAMPLE_TEXCUBE_LOD(unity_SpecCube0, r, _ReflectionMip);
                half3 probeSky = DecodeHDR(encodedSky, unity_SpecCube0_HDR);
                half3 skyGradient = lerp(_SkyHorizon.rgb, _SkyZenith.rgb, saturate(r.y));
                half skyAvailable = step(0.0001h, dot(probeSky, half3(0.2126h,0.7152h,0.0722h)));
                return lerp(skyGradient, probeSky, skyAvailable);
            }

            // głębokość wody (m) z tekstury dna (BraxWodaDno); poza prostokątem — wartość krawędzi (clamp)
            float Glebokosc(float2 xz)
            {
                float2 uv = (xz - _DnoProstokat.xy) / (_DnoProstokat.zw - _DnoProstokat.xy);
                float e = tex2Dlod(_DnoTex, float4(uv, 0.0, 0.0)).r;
                return e * e * _DnoMax;
            }

            v2f vert(appdata v)
            {
                v2f o;
                float3 world = mul(unity_ObjectToWorld,v.vertex).xyz;
                float2 phase = SwellPhase(world.xz);
                world.y += dot(sin(phase),float2(0.10,0.06)) * SwellEnvelope(world.xz).x;   // delikatne fale ok. 10 cm (31.10)
                o.world = world;
                o.position = mul(UNITY_MATRIX_VP,float4(world,1));
                UNITY_TRANSFER_FOG(o,o.position);
                return o;
            }

            // normalna lustra wody (w górę): długa fala + dwa kafle fal wiatrowych
            float3 NormalnaWody(float2 xz, float t)
            {
                // 1) długa fala (jak dotąd: te same fazy co łodzie, BraxVessel)
                float2 broadPhase = SwellPhase(xz);
                float3 envelope = SwellEnvelope(xz);
                float2 broad = cos(broadPhase) * float2(0.10*0.84,0.06*1.46);
                float height = dot(sin(broadPhase),float2(0.10,0.06));
                float2 slope = (broad.x * float2(0.84,0.5426)
                              + broad.y * float2(-0.4191,0.9079)) * envelope.x + height*envelope.yz;
                // 2) fale wiatrowe: dwa kafle mapy normalnych z widma (10 m i 23 m, obrót 36.9°), każdy płynie w swoją stronę;
                //    mipmapy wygładzają je z daleka (daleka woda = gładkie lustro nieba, jak na prawdziwym morzu)
                float2 uvA = xz / _FaleSkala + t * float2(0.031, 0.019);
                float2 xzB = float2(dot(xz, float2(0.8, 0.6)), dot(xz, float2(-0.6, 0.8)));
                float2 uvB = xzB / (_FaleSkala * 2.3) + t * float2(-0.012, 0.021);
                half3 nA = tex2D(_FaleNormal, uvA).rgb * 2.0h - 1.0h;
                half3 nB = tex2D(_FaleNormal, uvB).rgb * 2.0h - 1.0h;
                float2 sA = -nA.xy / max(nA.z, 0.25h);
                float2 sB = -nB.xy / max(nB.z, 0.25h);
                sB = float2(0.8 * sB.x - 0.6 * sB.y, 0.6 * sB.x + 0.8 * sB.y);
                slope += (sA * 0.7 + sB * 0.55) * _RippleStrength;
                return normalize(float3(-slope.x, 1.0, -slope.y));
            }

        ENDCG
        Pass
        {
            Tags { "LightMode"="ForwardBase" }
            ZWrite On
            Cull Back
            CGPROGRAM
            #pragma target 3.0
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile_fog
            half4 frag(v2f i) : SV_Target
            {
                // Full precision is intentional for world positions, time,
                // trigonometry and derivatives on mobile GLES3/WebGL2.
                float2 xz = i.world.xz;
                float t = _Time.y;
                float3 N = NormalnaWody(xz, t);

                float3 eye = _WorldSpaceCameraPos.xyz - i.world;
                float dist = max(length(eye), 0.001);
                float3 V = eye / dist;
                float NdotV = saturate(dot(N, V));
                float F = 0.02 + 0.98 * pow(1.0 - NdotV, 5.0);   // Fresnel (Schlick) dla wody n = 1.33

                // 3) odbite niebo
                float3 R = reflect(-V, N);
                float kat = max(length(fwidth(R)), 0.0008);
                half3 sky = Niebo(R, kat);

                // 4) światło sceny: słońce w czasie rzeczywistym, inaczej kierunek i barwa z materiału
                bool realne = dot(_LightColor0.rgb, half3(0.3h, 0.6h, 0.1h)) > 0.01h && _WorldSpaceLightPos0.w < 0.5;
                float3 L = realne ? normalize(_WorldSpaceLightPos0.xyz) : normalize(_SunDirection.xyz);
                half3 slonce = realne ? _LightColor0.rgb : _SlonceKolor.rgb;
                half3 otoczenie = dot(unity_AmbientSky.rgb, half3(1,1,1)) > 0.03h ? unity_AmbientSky.rgb : _Otoczenie.rgb;

                // 5) pod powierzchnią: światło rozproszone w wodzie + dno z piasku przez załamanie (płytka woda)
                half3 k = _Pochlanianie.rgb;
                half3 woda = _DeepColor.rgb;
                half3 pod = woda;
                float d0 = Glebokosc(xz);
                // rozmiar piksela na dnie liczony przed rozgałęzieniem (tex2Dgrad: bez pochodnych w warunku, bez migotania)
                float2 dX = ddx(xz), dY = ddy(xz);
                UNITY_BRANCH
                if (d0 < 9.0)
                {
                    float3 T = refract(-V, N, 0.7519);           // promień w wodzie (1 / 1.33)
                    float ty = max(-T.y, 0.15);
                    float2 hit = xz + T.xz * (d0 / ty);
                    float d1 = Glebokosc(hit);                   // dno w miejscu, gdzie promień je trafia
                    hit = xz + T.xz * (d1 / ty);
                    float path = d1 / ty;                        // droga od powierzchni do dna wzdłuż promienia
                    half3 albedo = tex2Dgrad(_DnoAlbedo, hit / _DnoSkala, dX / _DnoSkala, dY / _DnoSkala).rgb * _DnoKolor.rgb;
                    float2 cuv = hit / _KaustykiSkala, cX = dX / _KaustykiSkala, cY = dY / _KaustykiSkala;
                    half c1 = tex2Dgrad(_Kaustyki, cuv + t * float2(0.023, 0.011), cX, cY).r;
                    half c2 = tex2Dgrad(_Kaustyki, cuv * 1.37 + float2(0.31, 0.57) - t * float2(0.013, 0.027), cX * 1.37, cY * 1.37).r;
                    half kaust = min(c1 * c1, c2 * c2) * (4.0h / 0.755h);
                    kaust = lerp(1.0h, kaust, smoothstep(0.05, 0.4, d1) * exp(-d1 * 0.12));
                    float Ly = max(L.y, 0.05);
                    float sunPath = d1 / sqrt(1.0 - (1.0 - Ly * Ly) / 1.7689);   // promień słońca załamany w wodzie
                    half3 swiatloDna = slonce * (Ly * kaust) * exp(-k * sunPath) + otoczenie * exp(-k * d1);
                    half3 Tv = exp(-k * path);
                    pod = albedo * swiatloDna * Tv + woda * (1.0h - Tv);
                    pod = lerp(pod, woda, smoothstep(7.0, 9.0, d0));
                }

                // 6) Fresnel: światło spod powierzchni ↔ odbite niebo; odblask słońca (Blinn-Phong z normalizacją energii)
                half3 color = lerp(pod, sky, F * _ReflectionStrength);
                float3 Hv = normalize(L + V);
                float nh = saturate(dot(N, Hv));
                float p = lerp(1200.0, 180.0, saturate(dist / 400.0));
                float FH = 0.02 + 0.98 * pow(1.0 - saturate(dot(V, Hv)), 5.0);
                color += slonce * ((p + 8.0) / (8.0 * W_PI) * pow(nh, p) * FH * saturate(dot(N, L)));
                // 7) daleko (od 700 m): woda przechodzi w morze nieba miasta (MiastoTlo.cginc, tryb zdjęcia: kolor wody ze zdjęcia,
                //    tuż pod horyzontem samo zdjęcie) — koniec siatki morza za kamerą (1.6 km) bez widocznej krawędzi
                UNITY_BRANCH
                if (_DalekoDo > 0.0 && dist > _DalekoOd)
                {
                    float3 d = -V;
                    half cosv = saturate(-d.y);
                    half frd = 0.02h + 0.98h * pow(1.0h - cosv, 5.0h);
                    half3 dal = _DalekoKolor.rgb * lerp(0.78h, 1.0h, frd);
                    float3 rr = float3(d.x, abs(d.y), d.z);
                    dal += half3(1.0h, 0.96h, 0.88h) * pow(saturate(dot(rr, L)), 220.0) * 0.35h;
                    half wz = 1.0h - smoothstep(0.010, 0.040, -d.y);
                    if (wz > 0.001h) dal = lerp(dal, Zdjecie(d, kat), wz);
                    color = lerp(color, dal, smoothstep(_DalekoOd, _DalekoDo, dist));
                }
                half4 result = half4(BraxDobaWody(min(color, half3(8.0h, 8.0h, 8.0h))), 1);
                UNITY_APPLY_FOG(i.fogCoord,result);
                return result;
            }
            ENDCG
        }
        Pass
        {
            Tags { "LightMode"="ForwardBase" }
            ZWrite On
            Cull Front
            CGPROGRAM
            #pragma target 3.0
            #pragma vertex vert
            #pragma fragment fragPod
            #pragma multi_compile_fog
            // OD SPODU (kamera pod wodą — nurkowanie, BraxMiastoNurek): okno Snella — w stożku 48.8° od pionu (n = 1.33) niebo załamane na
            // falującej powierzchni, dalej całkowite odbicie wewnętrzne (światło rozproszone w głębi); po drodze od lustra do kamery
            // woda pochłania światło (te same współczynniki co z góry). Z góry ten przebieg nic nie rysuje (Cull Front).
            half4 fragPod(v2f i) : SV_Target
            {
                float2 xz = i.world.xz;
                float3 N = -NormalnaWody(xz, _Time.y);                  // normalna lustra od strony wody (w dół)
                float3 eye = _WorldSpaceCameraPos.xyz - i.world;
                float dist = max(length(eye), 0.001);
                float3 V = eye / dist;                                    // od lustra do kamery (w dół)
                float kat = max(length(fwidth(V)) * 4.0, 0.002);
                float3 T = refract(-V, N, 1.33);                          // promień z wody w powietrze (0 = całkowite odbicie)
                half3 k = _Pochlanianie.rgb;
                half3 glebia = _DeepColor.rgb * 1.6h;
                half3 color = glebia;
                UNITY_BRANCH
                if (dot(T, T) > 1e-4)
                {
                    half F = 0.02h + 0.98h * pow(1.0h - saturate(dot(-N, normalize(T))), 5.0h);
                    color = lerp(Niebo(T, kat), glebia, saturate(F));
                }
                half3 Tv = exp(-k * dist);
                color = color * Tv + glebia * (1.0h - Tv);
                half4 result = half4(BraxDobaWody(color), 1);
                UNITY_APPLY_FOG(i.fogCoord, result);
                return result;
            }
            ENDCG
        }
    }
    Fallback Off
}
